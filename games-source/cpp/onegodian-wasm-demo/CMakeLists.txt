cmake_minimum_required(VERSION 3.20)
project(onegodian_wasm_demo LANGUAGES CXX)
if(NOT EMSCRIPTEN)
 message(FATAL_ERROR "Requires Emscripten via emcmake")
endif()
add_executable(onegodian_wasm_demo src/main.cpp)
target_compile_features(onegodian_wasm_demo PRIVATE cxx_std_17)
target_link_options(onegodian_wasm_demo PRIVATE "-sWASM=1" "-sALLOW_MEMORY_GROWTH=1" "-sENVIRONMENT=web" "-sEXIT_RUNTIME=0" "--shell-file=${CMAKE_CURRENT_SOURCE_DIR}/shell.html")
set_target_properties(onegodian_wasm_demo PROPERTIES OUTPUT_NAME "index" SUFFIX ".html" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/dist")
